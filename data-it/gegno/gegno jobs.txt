mission "Gegno Asteroid Mining - Aluminum"
	job
	repeat
	name "Raccogli 10 tonnellate di alluminio"
	description "Estrai e raccogli 10 tonnellate di alluminio, poi riporta i minerali ai Gegno su <origin>."
	source "Tschyss"
	to offer
		has "Gegno Asteroid Mining Prologue: done"
		random < 10
	to accept
		"cargo space" > 10
	on visit
		dialog "Sei tornato su <origin>, ma non hai 10 tonnellate di alluminio nella stiva. Assicurati di estrarre 10 tonnellate di alluminio e di raccoglierle nel carico prima di atterrare."
	on complete
		"gegno: mining jobs" ++
		payment 36000
		outfit "Aluminum" -10
		dialog phrase "gegno mining payment dialog"
