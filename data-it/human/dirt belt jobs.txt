mission "Drought Relief"
	name `Soccorso per la siccità su <planet>`
	job
	repeat
	description `Il mondo agricolo di <destination> sta attraversando una siccità. Consegna <cargo> sul pianeta entro <date> per <payment>.`
	cargo "drought relief supplies" 25 2 .05
	deadline
	to offer
		random < 10
	source
		government "Republic"
		not attributes "farming"
	destination
		distance 3 10
		attributes "dirt belt"
		attributes "farming"
	on visit
		dialog phrase "generic cargo on visit"
	on complete
		payment 10000 160
		dialog "Scarichi il carico di <commodity> e ritiri il pagamento di <payment>. Un gruppo di cittadini di <planet> ti ringrazia e ti saluta."


mission "To Remembrance Day celebration [0]"
	name `Festa del Ricordo`
	job
	repeat
	description `Porta <fare> a <destination> entro <date> per la Festa del Ricordo del 19 febbraio. Il pagamento è <payment>.`
	passengers 4 4 .5
	deadline
	to offer
		or
			and
				random < 20
				month == 12
				day >= 21
			and
				random < 30
				month == 1
			and
				random < 10
				month == 2
				day < 19
	to fail
		month * 100 + day >= 219
	source
		attributes "dirt belt"
	destination
		distance 3 15
		attributes "dirt belt"
	on visit
		dialog phrase "generic passenger on visit"
	on complete
		payment 1000 180
		dialog "Auguri ai tuoi <passengers> per il loro soggiorno su <planet>. I bambini scendono dalla nave con la faccia dipinta di verde, e i genitori ti pagano <payment>."
