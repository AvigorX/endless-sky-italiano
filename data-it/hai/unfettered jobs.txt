mission "Unfettered Jump Drive 4"
	job
	repeat
	name "Vendi motore di salto"
	description "Scambia un motore di salto della tua ammiraglia o del carico con denaro, oppure con armi, se c'è un fornitore disponibile."
	source
		attributes "unfettered"
	to offer
		has "Unfettered Jump Drive 3: offered"
		not "event: wanderers: unfettered invasion starts"
	to fail
		has "Unfettered Jump Drive 4: active"
	to accept
		or
			has "outfit (flagship installed): Jump Drive"
			has "outfit (cargo): Jump Drive"
	on accept
		conversation
			branch "want outfits"
				not "unfettered: money for drives"
				has "Unfettered Jump Drive Trading: offered"
				has "flagship planet: Darkcloak"
			label money
			action
				payment 1000000
			`Come al solito, gli Unfettered sono più che disposti a pagarti un milione di crediti per il tuo motore di salto, ma parlando con loro non ottieni nessuna informazione in più.`
				goto trade

			label "want outfits"
			`L'accordo abituale degli Unfettered, scambiare un motore di salto con delle armi, è ancora valido.`
			choice
				`	(Prendi due Tripulse Shredders.)`
				`	(Prendi invece il denaro.)`
					goto money
			action
				outfit "Tripulse Shredder" 2
			`	I due Shredders vengono caricati sulla tua nave, in cambio del tuo motore di salto.`

			label trade
			branch spareJD
				"outfit (cargo): Jump Drive" + "outfit (flagship installed): Hyperdrive" + "outfit (flagship installed): Scram Drive" + "outfit (flagship installed): Jump Drive" > 1
			action
				outfit "Hyperdrive" 1
			label spareJD
			action
				outfit "Jump Drive" -1
				"reputation: Hai (Unfettered)" >?= 40
